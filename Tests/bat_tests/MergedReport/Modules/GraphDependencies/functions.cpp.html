void parseSingleLine(const QString& rawLine, int lineNumber, QList<Action*>& actions, QSet<Error>& errors)
{

    const QString trimmedLine = rawLine.trimmed();

    if (action->targetRoot != nullptr && action->expression != nullptr) {

    }
}

// ÐÐ¾ÑÑÑÐ¾ÐµÐ½Ð¸Ðµ Ð´ÐµÑÐµÐ²Ð° Ð²ÑÑÐ°Ð¶ÐµÐ½Ð¸Ñ Ð¸Ð· Ð¿Ð¾ÑÑÑÐ¸ÐºÑÐ½Ð¾Ð¹ ÑÐ¾ÑÐ¼Ñ
ExprNode* parseExpression(const QString& exprStr,

                          QSet<Error>& errors)
{
    // Ð Ð°Ð·Ð±Ð¸ÑÑ ÑÐµÐºÑÑ Ð¿Ð¾ Ð¿ÑÐ¾Ð±ÐµÐ»Ð°Ð¼
    const QStringList tokens = exprStr.split(' ', Qt::SkipEmptyParts);

    }

        result = stack.pop();

        // Ð¡ÑÐµÐº Ð¿ÑÑÑ - ÐºÐ¾ÑÐ½Ñ Ð½ÐµÑ, Ð·Ð°ÑÐ¸ÐºÑÐ¸ÑÐ¾Ð²Ð°ÑÑ Ð¾ÑÐ¸Ð±ÐºÑ
        errors.insert(Error(ExtraOperands, lineNumber, 0, ""));

// ÐÐ»Ð°ÑÑÐ¸ÑÐ¸ÐºÐ°ÑÐ¸Ñ Ð½ÐµÐºÐ¾ÑÑÐµÐºÑÐ½Ð¾Ð³Ð¾ ÑÐ¾ÐºÐµÐ½Ð°: ÑÐ¾ÑÐ½ÑÐ¹ ÑÐ¸Ð¿ Ð¾ÑÐ¸Ð±ÐºÐ¸
ErrorType classifyInvalidToken(const QString& token)
{
    // ÐÑÑÑÐ¾Ð¹ ÑÐ¾ÐºÐµÐ½ - Ð½ÐµÐ¸Ð·Ð²ÐµÑÑÐ½ÑÐ¹ Ð¾Ð¿ÐµÑÐ°ÑÐ¾Ñ
    if (token.isEmpty()) {
        return UnsupportedOperator;
    }
    const QChar firstChar = token[0];
    // ÐÐ¿ÑÐµÐ´ÐµÐ»Ð¸ÑÑ, Ð¿Ð¾ÑÐ¾Ð¶ Ð»Ð¸ ÑÐ¾ÐºÐµÐ½ Ð½Ð° ÑÐ¸ÑÐ»Ð¾
    const bool looksNumeric = firstChar.isDigit() ||
                              (token.size() > 1 && firstChar == '-' && token[1].isDigit());
    // ÐÐµ Ð¿Ð¾ÑÐ¾Ð¶ Ð½Ð° ÑÐ¸ÑÐ»Ð¾ - Ð½ÐµÐ¸Ð·Ð²ÐµÑÑÐ½ÑÐ¹ Ð¾Ð¿ÐµÑÐ°ÑÐ¾Ñ/ÑÐ¸Ð¼Ð²Ð¾Ð»
    if (!looksNumeric) {
        return UnsupportedOperator;
    }
    // ÐÐ¾ÑÐ¾Ð¶ Ð½Ð° ÑÐ¸ÑÐ»Ð¾: ÐµÑÐ»Ð¸ ÑÐ¾Ð»ÑÐºÐ¾ ÑÐ¸ÑÑÑ/ÑÐ¾ÑÐºÐ°/Ð¼Ð¸Ð½ÑÑ - Ð½ÐµÐ²ÐµÑÐ½ÑÐ¹ ÑÐ¾ÑÐ¼Ð°Ñ ÑÐ¸ÑÐ»Ð°, Ð¸Ð½Ð°ÑÐµ Ð¸Ð¼Ñ Ñ Ð½ÐµÐ´Ð¾Ð¿ÑÑÑÐ¸Ð¼ÑÐ¼Ð¸ ÑÐ¸Ð¼Ð²Ð¾Ð»Ð°Ð¼Ð¸
    const QString numericChars = "0123456789.-";
    for (const QChar c : token) {
        if (!numericChars.contains(c)) {
            return InvalidVariableName;
        }
    }
    return InvalidNumberFormat;
}

// Ð Ð°Ð·Ð±Ð¾Ñ Ð¾Ð´Ð½Ð¾Ð³Ð¾ ÑÐ¾ÐºÐµÐ½Ð° (Ð´Ð¸ÑÐ¿ÐµÑÑÐµÑ)
void parseSingleToken(const QString& token,
                      QStack<ExprNode*>& stack,
                      int lineNumber,
                      int tokenIndex,
                      const QRegularExpression& numRegex,
                      const QRegularExpression& varRegex,
                      QSet<Error>& errors)
{
    // ÐÐ¸Ð½Ð°ÑÐ½ÑÐ¹ Ð¾Ð¿ÐµÑÐ°ÑÐ¾Ñ (+ - * /)
    if (isBinaryOperator(token)) {
        handleBinaryOp(token, stack, lineNumber, tokenIndex, errors);
    }
    // Ð£Ð½Ð°ÑÐ½ÑÐ¹ Ð¾Ð¿ÐµÑÐ°ÑÐ¾Ñ (++ --)
    else if (token == "++" || token == "--") {
        handleUnaryOp(token, stack, lineNumber, tokenIndex, errors);
    }
    // ÐÐ¾ÑÑÑÐ¿ Ðº ÑÐ»ÐµÐ¼ÐµÐ½ÑÑ Ð¼Ð°ÑÑÐ¸Ð²Ð° ([])
    else if (token == "[]") {
        handleArrayAccess(token, stack, lineNumber, tokenIndex, errors);
    }
    // Ð¦ÐµÐ»Ð¾Ðµ ÑÐ¸ÑÐ»Ð¾
    else if (numRegex.match(token).hasMatch()) {
        handleNumberToken(token, stack, lineNumber, tokenIndex, errors);
    }
    // ÐÐ¼Ñ Ð¿ÐµÑÐµÐ¼ÐµÐ½Ð½Ð¾Ð¹
    else if (varRegex.match(token).hasMatch()) {
        handleVariableToken(token, stack, lineNumber, tokenIndex, errors);
    }
    // ÐÐµÐºÐ¾ÑÑÐµÐºÑÐ½ÑÐ¹ ÑÐ¾ÐºÐµÐ½: Ð¾Ð¿ÑÐµÐ´ÐµÐ»Ð¸ÑÑ ÑÐ¾ÑÐ½ÑÐ¹ Ð²Ð¸Ð´ Ð¾ÑÐ¸Ð±ÐºÐ¸
    else {
        errors.insert(Error(classifyInvalidToken(token), lineNumber, tokenIndex, token));
        stack.push(makeStubNode());
    }
}

// ÐÐ±ÑÐ°Ð±Ð¾ÑÑÐ¸Ðº Ð±Ð¸Ð½Ð°ÑÐ½Ð¾Ð³Ð¾ Ð¾Ð¿ÐµÑÐ°ÑÐ¾ÑÐ°
void handleBinaryOp(const QString& token, QStack<ExprNode*>& stack, int lineNumber, int tokenIndex, QSet<Error>& errors)
{

    if (stack.size() < 2) {
        errors.insert(Error(NotEnoughOperands, lineNumber, tokenIndex, token));
        // ÐÐ¾Ð»Ð¾Ð¶Ð¸ÑÑ Ð·Ð°Ð³Ð»ÑÑÐºÑ Ð¸ Ð²ÑÐ¹ÑÐ¸
        stack.push(makeStubNode());
        return;
    }
    // Ð¡Ð½ÑÑÑ Ñ Ð²ÐµÑÑÐ¸Ð½Ñ Ð¿ÑÐ°Ð²ÑÐ¹ Ð¾Ð¿ÐµÑÐ°Ð½Ð´
    ExprNode* right = stack.pop();
    // Ð¡Ð½ÑÑÑ Ñ Ð²ÐµÑÑÐ¸Ð½Ñ Ð»ÐµÐ²ÑÐ¹ Ð¾Ð¿ÐµÑÐ°Ð½Ð´
    ExprNode* left  = stack.pop();
    // ÐÐ¿ÑÐµÐ´ÐµÐ»Ð¸ÑÑ Ð¾Ð¿ÐµÑÐ°ÑÐ¸Ñ
    TokenType opType = Plus;
    if (token == "-") opType = Minus;
    else if (token == "*") opType = Multiply;
    else if (token == "/") opType = Divide;
    // Ð¡Ð¾Ð·Ð´Ð°ÑÑ ÑÐ·ÐµÐ» Ð¾Ð¿ÐµÑÐ°ÑÐ¸Ð¸
    ExprNode* opNode = new ExprNode(opType);
    // ÐÐ¾Ð¼ÐµÑÑÐ¸ÑÑ Ð»ÐµÐ²ÑÐ¹ Ð¸ Ð¿ÑÐ°Ð²ÑÐ¹ Ð¾Ð¿ÐµÑÐ°Ð½Ð´Ñ
    opNode->leftOperand  = left;
    opNode->rightOperand = right;
    // ÐÐµÑÐ½ÑÑÑ Ð² ÑÑÐµÐº
    stack.push(opNode);
}

// ÐÐ±ÑÐ°Ð±Ð¾ÑÑÐ¸Ðº ÑÐ½Ð°ÑÐ½Ð¾Ð³Ð¾ Ð¾Ð¿ÐµÑÐ°ÑÐ¾ÑÐ°
void handleUnaryOp(const QString& token, QStack<ExprNode*>& stack, int lineNumber, int tokenIndex, QSet<Error>& errors)
{

    if (stack.size() < 1) {

        // ÐÐ¾Ð»Ð¾Ð¶Ð¸ÑÑ Ð·Ð°Ð³Ð»ÑÑÐºÑ Ð¸ Ð²ÑÐ¹ÑÐ¸
        stack.push(makeStubNode());
        return;
    }
    // Ð¡Ð½ÑÑÑ Ð¾Ð¿ÐµÑÐ°Ð½Ð´ Ñ Ð²ÐµÑÑÐ¸Ð½Ñ ÑÑÐµÐºÐ°
    ExprNode* operand = stack.pop();

    if (operand->type != Var) {
        errors.insert(Error(UnaryOperandNotVariable, lineNumber, tokenIndex, token));
    }
    // ÐÐ¿ÑÐµÐ´ÐµÐ»Ð¸ÑÑ Ð¾Ð¿ÐµÑÐ°ÑÐ¸Ñ
    TokenType opType = (token == "++") ? Increment : Decrement;
    // Ð¡Ð¾Ð·Ð´Ð°ÑÑ ÑÐ·ÐµÐ»
    ExprNode* opNode = new ExprNode(opType);
    // ÐÐ¾Ð¼ÐµÑÑÐ¸ÑÑ Ð¾Ð¿ÐµÑÐ°Ð½Ð´ Ð»ÐµÐ²ÑÐ¼ Ð¿Ð¾ÑÐ¾Ð¼ÐºÐ¾Ð¼
    opNode->leftOperand = operand;
    // ÐÐ¾Ð»Ð¾Ð¶Ð¸ÑÑ Ð² ÑÑÐµÐº
    stack.push(opNode);
}

DependencyType compareIndices(ExprNode* node1, ExprNode* node2)
{
    DependencyType result;

    if (!node1 && !node2) {
        // ÐÑÐ»Ð¸ Ð¾Ð±Ð° ÑÐ·Ð»Ð° Ð¿ÑÑÑÑ, Ð²ÐµÑÐ½ÑÑÑ Direct
        result = Direct;
    } else if (!node1 || !node2) {
        // ÐÑÐ»Ð¸ Ð¿ÑÑÑ ÑÐ¾Ð²Ð½Ð¾ Ð¾Ð´Ð¸Ð½, Ð²ÐµÑÐ½ÑÑÑ General
        result = General;
    } else if (node1->type != node2->type) {
        // ÐÑÐ»Ð¸ ÑÐ¸Ð¿Ñ ÑÐ·Ð»Ð¾Ð² ÑÐ°Ð·Ð»Ð¸ÑÐ°ÑÑÑÑ, Ð²ÐµÑÐ½ÑÑÑ General
        result = General;

        // ÐÑÐ»Ð¸ ÑÐ·Ð»Ñ = Ð¿ÐµÑÐµÐ¼ÐµÐ½Ð½ÑÐµ: Ð¸Ð¼ÐµÐ½Ð° ÑÐ°Ð²Ð½Ñ - Direct, Ð¸Ð½Ð°ÑÐµ General
        result = (node1->value == node2->value) ? Direct : General;
    } else if (node1->type == Number) {
        // ÐÑÐ»Ð¸ ÑÐ·Ð»Ñ = ÑÐ¸ÑÐ»Ð°: Ð·Ð½Ð°ÑÐµÐ½Ð¸Ñ ÑÐ°Ð²Ð½Ñ - Direct, Ð¸Ð½Ð°ÑÐµ NoDependency
        result = (node1->value == node2->value) ? Direct : NoDependency;
    } else if (node1->type == ArrayAccess) {
        // ÐÑÐ»Ð¸ ÑÐ·Ð»Ñ - Ð´Ð¾ÑÑÑÐ¿ Ðº Ð¼Ð°ÑÑÐ¸Ð²Ñ: ÑÑÐ°Ð²Ð½Ð¸ÑÑ Ð±Ð°Ð·Ñ Ð¸ Ð¸Ð½Ð´ÐµÐºÑÑ ÑÐµÐºÑÑÑÐ¸Ð²Ð½Ð¾
        const DependencyType leftResult  = compareIndices(node1->leftOperand,  node2->leftOperand);
        const DependencyType rightResult = compareIndices(node1->rightOperand, node2->rightOperand);
        // ÐÑÐ»Ð¸ ÑÐ¾ÑÑ Ð±Ñ Ð¾Ð´Ð¸Ð½ ÑÐµÐ·ÑÐ»ÑÑÐ°Ñ NoDependency - Ð²ÐµÑÐ½ÑÑÑ NoDependency
        if (leftResult == NoDependency || rightResult == NoDependency) {
            result = NoDependency;
            // ÐÐ½Ð°ÑÐµ ÐµÑÐ»Ð¸ ÑÐ¾ÑÑ Ð±Ñ Ð¾Ð´Ð¸Ð½ General - Ð²ÐµÑÐ½ÑÑÑ General
        } else if (leftResult == General || rightResult == General) {
            result = General;
            // ÐÐ½Ð°ÑÐµ - Ð²ÐµÑÐ½ÑÑÑ Direct
        } else {
            result = Direct;
        }
    } else {
        result = General;
    }

    return result;
}
