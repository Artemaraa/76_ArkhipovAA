// ÐÑÐ¾Ð²ÐµÑÐºÐ° ÑÐ°Ð·Ð¼ÐµÑÐ½Ð¾ÑÑÐµÐ¹ Ð¿ÐµÑÐµÐ¼ÐµÐ½Ð½ÑÑ Ð´ÐµÐ¹ÑÑÐ²Ð¸Ñ; Ð²Ð¾Ð·Ð²ÑÐ°ÑÐ°ÐµÑ true Ð¿ÑÐ¸ Ð½Ð°Ð»Ð¸ÑÐ¸Ð¸ Ð¾ÑÐ¸Ð±ÐºÐ¸
bool DependencyGraph::checkActionDimensions(Action* currentAction, QMap<QString, int>& dims, QSet<Error>& errors)
{
    // Ð¡Ð¾Ð±ÑÐ°ÑÑ Ð¿ÐµÑÐµÐ¼ÐµÐ½Ð½ÑÐµ Ð»ÐµÐ²Ð¾Ð¹ ÑÐ°ÑÑÐ¸ (ÑÐµÐ»Ñ) Ð¸ Ð¿ÑÐ°Ð²Ð¾Ð¹ ÑÐ°ÑÑÐ¸ (Ð¸ÑÑÐ¾ÑÐ½Ð¸ÐºÐ¸)
    QList<ExprNode*> vars;
    collectSources(currentAction->targetRoot, vars);
    vars.append(currentAction->sourceVariables);
    // ÐÑÐ¾Ð²ÐµÑÐ¸ÑÑ ÐºÐ°Ð¶Ð´ÑÑ Ð¿ÐµÑÐµÐ¼ÐµÐ½Ð½ÑÑ; Ð²ÑÐµ Ð¾ÑÐ¸Ð±ÐºÐ¸ ÐºÐ¾Ð¿Ð¸Ð¼, Ð½Ðµ Ð²ÑÑÐ¾Ð´Ð¸Ð¼ Ð½Ð° Ð¿ÐµÑÐ²Ð¾Ð¹
    bool hasDimError = false;
    for (ExprNode* node : vars) {
        const QString name = getArrayName(node);
        if (name.isEmpty()) {
            continue;
        }
        Error dimError;
        if (!validateArrayDimension(node, name, dims, currentAction->number, dimError)) {
            errors.insert(dimError);
            hasDimError = true;
        }
    }
    return hasDimError;
}

// ÐÐ¾ÑÑÑÐ¾ÐµÐ½Ð¸Ðµ ÑÑÐ±ÐµÑ Ð·Ð°Ð²Ð¸ÑÐ¸Ð¼Ð¾ÑÑÐ¸ Ð´Ð»Ñ Ð¿ÑÐ¾ÑÐ¸ÑÐ°Ð½Ð½ÑÑ Ð¿ÐµÑÐµÐ¼ÐµÐ½Ð½ÑÑ
void DependencyGraph::buildEdgesForAction(Action* currentAction, const QList<ExprNode*>& sourceVariables, const QMap<QString, Action*>& varTable)
{
    for (ExprNode* varNode : sourceVariables) {
        Action* dependencyAction = nullptr;
        const DependencyType depType = determineDependency(varNode, varTable, dependencyAction);
        // ÐÑÐ»Ð¸ Ð·Ð°Ð²Ð¸ÑÐ¸Ð¼Ð¾ÑÑÑ ÐµÑÑÑ, Ð´Ð¾Ð±Ð°Ð²Ð¸ÑÑ ÑÐµÐ±ÑÐ¾ Ð¾Ñ ÑÐµÐºÑÑÐµÐ³Ð¾ Ð´ÐµÐ¹ÑÑÐ²Ð¸Ñ Ðº Ð½Ð°Ð¹Ð´ÐµÐ½Ð½Ð¾Ð¼Ñ
        if (depType != NoDependency && dependencyAction != nullptr) {
            addEdge(currentAction, dependencyAction, depType);
        }
    }
}

void DependencyGraph::updateVarTable(Action* currentAction, const QString& targetName, const QList<ExprNode*>& modifiedVariables, QMap<QString, Action*>& varTable)

    varTable[targetName] = currentAction;
    // ÐÐ»Ñ ÐºÐ°Ð¶Ð´Ð¾Ð¹ Ð¸Ð·Ð¼ÐµÐ½ÑÐµÐ¼Ð¾Ð¹ Ð¿ÐµÑÐµÐ¼ÐµÐ½Ð½Ð¾Ð¹ (++/--) Ð¾Ð±Ð½Ð¾Ð²Ð¸ÑÑ Ð·Ð°Ð¿Ð¸ÑÑ Ð½Ð° ÑÐµÐºÑÑÐµÐµ Ð´ÐµÐ¹ÑÑÐ²Ð¸Ðµ

            if (!varName.isEmpty()) {
                varTable[varName] = currentAction;
            }
        }
    }
}

bool DependencyGraph::validateArrayDimension(ExprNode* node,
                                             const QString& varName,
                                             QMap<QString, int>& dims,
                                             int lineNumber,
                                             Error& error)
{
    // Ð Ð°Ð·Ð¼ÐµÑÐ½Ð¾ÑÑÑ ÑÑÐ¾Ð¹ Ð¿ÐµÑÐµÐ¼ÐµÐ½Ð½Ð¾Ð¹ (0 = ÑÐºÐ°Ð»ÑÑ, >=1 = Ð¼Ð°ÑÑÐ¸Ð²)

    if (!dims.contains(varName)) {
        dims[varName] = currentDim;
        return true;
    }
    // Ð¡Ð²ÐµÑÐ¸ÑÑ Ñ Ð·Ð°ÑÐ¸ÐºÑÐ¸ÑÐ¾Ð²Ð°Ð½Ð½Ð¾Ð¹ ÑÐ°Ð½ÐµÐµ ÑÐ°Ð·Ð¼ÐµÑÐ½Ð¾ÑÑÑÑ
    const int savedDim = dims[varName];
    if (currentDim == savedDim) {
        return true;
    }
    // Ð Ð°Ð·Ð¼ÐµÑÐ½Ð¾ÑÑÐ¸ ÑÐ°Ð·Ð¾ÑÐ»Ð¸ÑÑ - Ð²ÑÐ±ÑÐ°ÑÑ ÑÐ¾ÑÐ½ÑÐ¹ ÑÐ¸Ð¿ Ð¾ÑÐ¸Ð±ÐºÐ¸
    if (savedDim == 0 && currentDim > 0) {
        error = Error(ScalarWithIndex, lineNumber, 0, varName);      // Ð±ÑÐ» ÑÐºÐ°Ð»ÑÑ, ÑÑÐ°Ð» Ð¼Ð°ÑÑÐ¸Ð²
    } else if (savedDim > 0 && currentDim == 0) {
        error = Error(ArrayWithoutIndex, lineNumber, 0, varName);    // Ð±ÑÐ» Ð¼Ð°ÑÑÐ¸Ð², ÑÑÐ°Ð» ÑÐºÐ°Ð»ÑÑ
    } else {
        error = Error(InvalidArrayDimension, lineNumber, 0, varName); // Ð´ÑÑÐ³Ð¾Ðµ ÑÐ¸ÑÐ»Ð¾ Ð¸Ð·Ð¼ÐµÑÐµÐ½Ð¸Ð¹
    }
    return false;
}
